<script>
// Initialize Studio on page load
let studio = null;

async function initializeStudio() {
  studio = await initStudio();
  
  // Register UI callbacks
  studio.registerUICallback('beatLoaded', (beatData) => {
    document.getElementById('beatName').textContent = beatData.name;
    document.getElementById('beatBPM').textContent = beatData.bpm;
    document.getElementById('beatKey').textContent = beatData.key;
    document.getElementById('beatGenre').textContent = beatData.genre;
  });

  studio.registerUICallback('beatPlaying', () => {
    document.getElementById('beatPlayBtn').classList.add('active');
    updateBeatProgress();
  });

  studio.registerUICallback('beatPaused', () => {
    document.getElementById('beatPlayBtn').classList.remove('active');
  });

  studio.registerUICallback('recordingStarted', () => {
    document.getElementById('recordBtn').classList.add('recording');
    document.getElementById('recordStatus').textContent = 'En cours...';
    document.getElementById('recordTimer').style.display = 'block';
    updateRecordingTimer();
  });

  studio.registerUICallback('recordingStopped', () => {
    document.getElementById('recordBtn').classList.remove('recording');
    document.getElementById('recordStatus').textContent = 'Enregistrement terminé';
    document.getElementById('recordTimer').style.display = 'none';
  });

  // Load sample beats
  loadBeats();
  
  console.log('✅ Studio UI initialized');
}

function loadBeats() {
  // Fetch beats from Firebase
  db.collection('beats').limit(6).get().then(snapshot => {
    const selector = document.getElementById('beatSelector');
    selector.innerHTML = '';

    snapshot.forEach(doc => {
      const beat = doc.data();
      const beatBtn = document.createElement('button');
      beatBtn.className = 'control-btn';
      beatBtn.innerHTML = `
        <div style="font-size: 1.5rem; margin-bottom: 4px;">🎵</div>
        <div style="font-size: 0.6rem; font-weight: bold;">${beat.title}</div>
      `;
      beatBtn.onclick = () => selectBeat(doc.id, beat);
      selector.appendChild(beatBtn);
    });
  });
}

async function selectBeat(id, beatData) {
  try {
    // Get download URL from Firebase Storage
    const storage = firebase.storage();
    const beatRef = storage.ref(`beats/${beatData.audioFile}`);
    const url = await beatRef.getDownloadURL();
    
    await studio.loadBeatFromURL(url, {
      name: beatData.title,
      bpm: beatData.bpm || 120,
      key: beatData.key || 'C',
      genre: beatData.genre || 'Hip-Hop'
    });
    
    document.getElementById('beatDuration').textContent = formatTime(studio.beatPlayer.getDuration());
  } catch (error) {
    console.error('Error loading beat:', error);
    showToast('Error loading beat');
  }
}

// Beat Player Controls
function studioBeatPlay() {
  studio.playBeat();
}

function studioBeatPause() {
  studio.pauseBeat();
}

function studioBeatStop() {
  studio.stopBeat();
}

function seekBeat(event) {
  const rect = event.currentTarget.getBoundingClientRect();
  const x = event.clientX - rect.left;
  const percentage = x / rect.width;
  const duration = studio.beatPlayer.getDuration();
  studio.seekBeat(duration * percentage);
}

function updateBeatProgress() {
  const interval = setInterval(() => {
    if (!studio.beatPlayer.isPlaying) {
      clearInterval(interval);
      return;
    }

    const progress = studio.getBeatProgress();
    document.getElementById('beatProgressFill').style.width = progress.percentage + '%';
    document.getElementById('beatCurrentTime').textContent = formatTime(progress.current);
  }, 100);
}

// Recording Controls
function toggleRecording() {
  if (studio.vocalRecorder.isRecording) {
    studio.stopRecording();
  } else {
    studio.startRecording();
  }
}

function updateRecordingTimer() {
  const interval = setInterval(() => {
    if (!studio.vocalRecorder.isRecording) {
      clearInterval(interval);
      return;
    }

    const duration = studio.getRecordingDuration() / 1000;
    document.getElementById('recordTimer').textContent = formatTime(duration);
    
    // Update mic level
    const level = Math.min(100, studio.getMicLevel() * 200);
    document.getElementById('micLevelBar').style.width = level + '%';
  }, 50);
}

// Mixer Controls
function updateBeatVolume(value) {
  studio.setBeatVolume(value);
  document.getElementById('beatVolValue').textContent = value + '%';
}

function updateVocalVolume(value) {
  studio.setVocalVolume(value);
  document.getElementById('vocalVolValue').textContent = value + '%';
}

function toggleEffect(effect) {
  const btn = document.getElementById(effect + 'Btn');
  btn.classList.toggle('active');

  if (effect === 'reverb') {
    studio.setReverb(btn.classList.contains('active') ? 50 : 0);
  } else if (effect === 'delay') {
    studio.setDelay(btn.classList.contains('active') ? 50 : 0);
  }
}

function toggleMute() {
  // Implement mute logic
}

function toggleSolo() {
  // Implement solo logic
}

// EQ Controls
function updateEQ(band, value) {
  const dB = (value - 0) / 10;
  const bandIndex = band === 'bass' ? 0 : band === 'mid' ? 1 : 2;
  
  studio.setEQ(bandIndex, value);
  
  document.getElementById('eq' + band.charAt(0).toUpperCase() + band.slice(1) + 'Value').textContent = dB.toFixed(1) + ' dB';
}

// Export & Publish
async function exportFreestyle() {
  showToast('Préparation de l\'export...');
  
  try {
    const wavBlob = await studio.exportRecording();
    
    if (!wavBlob) {
      showToast('Aucun enregistrement à exporter');
      return;
    }

    // Convert to MP3 or download as WAV
    const url = URL.createObjectURL(wavBlob);
    const link = document.createElement('a');
    link.href = url;
    link.download = `freestyle_${Date.now()}.wav`;
    link.click();
    URL.revokeObjectURL(url);
    
    showToast('✅ Freestyle exporté avec succès');
  } catch (error) {
    console.error('Export error:', error);
    showToast('Erreur lors de l\'export');
  }
}

async function publishFreestyle() {
  showToast('Publication en cours...');
  
  try {
    const user = auth.currentUser;
    if (!user) {
      showToast('Connecte-toi d\'abord');
      return;
    }

    const wavBlob = await studio.exportRecording();
    const storageRef = firebase.storage().ref(`freestyles/${user.uid}/${Date.now()}.wav`);
    
    await storageRef.put(wavBlob);
    const url = await storageRef.getDownloadURL();

    // Save metadata to Firestore
    await db.collection('freestyles').add({
      userId: user.uid,
      title: `Freestyle - ${new Date().toLocaleDateString()}`,
      audioUrl: url,
      createdAt: new Date(),
      likes: 0,
      plays: 0
    });

    showToast('✅ Freestyle publié sur ton profil');
  } catch (error) {
    console.error('Publish error:', error);
    showToast('Erreur lors de la publication');
  }
}

function shareFreestyle() {
  const link = window.location.href + '#freestyle';
  navigator.clipboard.writeText(link);
  showToast('✅ Lien copié');
}

// Initialize when Studio page is shown
document.addEventListener('studioReady', initializeStudio);

// Manual init
if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', initializeStudio);
} else {
  initializeStudio();
}
</script>
